From the terminal

# 400 Bad Request
printf 'GARBAGE\r\n\r\n' | nc localhost 8080

# 405 Method Not Allowed (method not in allow_methods for the location)
curl -i -X DELETE http://localhost:8080/index.html

# 411 Length Required
printf 'POST /uploads/ HTTP/1.0\r\n\r\n' | nc localhost 8080

# 413 Payload Too Large
head -c 20000000 /dev/zero | curl -i -X POST --data-binary @- http://localhost:8080/uploads/

# 414 URI Too Long
curl -i "http://localhost:8080/$(head -c 10000 /dev/zero | tr '\0' a)"

# 501 Not Implemented
curl -i -X PATCH http://localhost:8080/

# 505 HTTP Version Not Supported
printf 'GET / HTTP/2.0\r\n\r\n' | nc localhost 8080
